cmake_minimum_required(VERSION 3.22.1)
project(AppForgeRustJsi LANGUAGES CXX)
find_package(ReactAndroid REQUIRED CONFIG)
add_library(appforge_jsi SHARED IMPORTED)
set_target_properties(appforge_jsi PROPERTIES IMPORTED_LOCATION "${CMAKE_CURRENT_SOURCE_DIR}/../jniLibs/${ANDROID_ABI}/libappforge_jsi.so")
add_library(appforge_rust_jsi SHARED RustJsi.cpp)
target_compile_features(appforge_rust_jsi PRIVATE cxx_std_20)
target_link_libraries(appforge_rust_jsi PRIVATE ReactAndroid::jsi appforge_jsi)
target_link_options(appforge_rust_jsi PRIVATE "-Wl,-z,max-page-size=16384" "-Wl,-z,common-page-size=16384")
